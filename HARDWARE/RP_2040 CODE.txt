// -----------------------------------------------------------------------------
// FFT SIMPLE IMPLEMENTATION
// -----------------------------------------------------------------------------
void simpleFFT(float* input, float* output, int n) {
  for (int k = 0; k < n/2; k++) {
    float real = 0;
    float imag = 0;
    
    for (int t = 0; t < n; t++) {
      float angle = 2 * PI * k * t / n;
      real += input[t] * cos(angle);
      imag += input[t] * sin(angle);
    }
    output[k] = sqrt(real*real + imag*imag);
  }
}

// -----------------------------------------------------------------------------
// CALCULATE AUDIO FEATURES (INDUSTRIAL-GRADE - 16 features)
// -----------------------------------------------------------------------------
void calculateAudioFeatures() {
  if (samplesRead <= 0) return;
  
  int numSamples = min(samplesRead, fftSize);
  
  float sumSquares = 0;
  float peak = 0;
  
  // Calculate basic mic level (RMS of raw samples)
  long rawSum = 0;
  for (int i = 0; i < samplesRead; i++) {
    rawSum += (long)sampleBuffer[i] * sampleBuffer[i];
  }
  micLevel = sqrt(rawSum / (float)samplesRead);
  
  // Normalize and prepare FFT buffer
  for (int i = 0; i < numSamples; i++) {
    float sample = sampleBuffer[i] / 32768.0f;
    fftBuffer[i] = sample;
    sumSquares += sample * sample;
    
    float absSample = sample < 0 ? -sample : sample;
    if (absSample > peak) peak = absSample;
  }
  
  // Basic time-domain features
  audioRMS = sqrt(sumSquares / (float)numSamples);
  audioPeak = peak;
  audioEnergy = sumSquares;
  audioCrestFactor = audioRMS > 0 ? audioPeak / audioRMS : 0;
  
  // Frequency analysis using simple FFT
  float spectrum[fftSize/2];  // 128 bins
  simpleFFT(fftBuffer, spectrum, numSamples);
  
  // Spectral statistics
  float spectralSum = 0;
  float spectralWeightedSum = 0;
  float logSum = 0;
  float totalEnergy = 0;
  
  for (int i = 0; i < fftSize/2; i++) {
    spectralWeightedSum += spectrum[i] * i;
    spectralSum += spectrum[i];
    totalEnergy += spectrum[i] * spectrum[i];
    logSum += log(spectrum[i] + 1e-10f);
  }
  
  // Spectral Centroid
  audioSpectralCentroid = spectralSum > 0 ? spectralWeightedSum / spectralSum : 0;
  
  // Spectral Flatness (geometric mean / arithmetic mean)
  float arithmeticMean = spectralSum / (fftSize/2);
  float geometricMean = exp(logSum / (fftSize/2));
  audioSpectralFlatness = arithmeticMean > 0 ? geometricMean / arithmeticMean : 0;
  
  // Spectral Bandwidth (weighted std deviation from centroid)
  float bandwidthSum = 0;
  for (int i = 0; i < fftSize/2; i++) {
    float diff = i - audioSpectralCentroid;
    bandwidthSum += spectrum[i] * diff * diff;
  }
  audioSpectralBandwidth = spectralSum > 0 ? sqrt(bandwidthSum / spectralSum) : 0;
  
  // Spectral Rolloff (frequency where 85% of energy accumulates)
  float energyThreshold = totalEnergy * 0.85f;
  float cumulativeEnergy = 0;
  audioSpectralRolloff = fftSize/2 - 1;
  for (int i = 0; i < fftSize/2; i++) {
    cumulativeEnergy += spectrum[i] * spectrum[i];
    if (cumulativeEnergy >= energyThreshold) {
      audioSpectralRolloff = i;
      break;
    }
  }
  
  // MFCC with 8 mel bands and 8 coefficients
  float melBands[8] = {0};
  for (int i = 0; i < 8; i++) {
    int startBin = i * (fftSize/16);
    int endBin = (i + 1) * (fftSize/16);
    for (int j = startBin; j < endBin; j++) {
      melBands[i] += spectrum[j];
    }
    melBands[i] = log(melBands[i] + 1e-6f);
  }
  
  // DCT for 8 MFCC coefficients
  float mfccCoeffs[8] = {0};
  for (int i = 0; i < 8; i++) {
    for (int j = 0; j < 8; j++) {
      mfccCoeffs[i] += melBands[j] * cos(PI * i * (j + 0.5f) / 8);
    }
  }
  
  audioMFCC1 = mfccCoeffs[0];
  audioMFCC2 = mfccCoeffs[1];
  audioMFCC3 = mfccCoeffs[2];
  audioMFCC4 = mfccCoeffs[3];
  audioMFCC5 = mfccCoeffs[4];
  audioMFCC6 = mfccCoeffs[5];
  audioMFCC7 = mfccCoeffs[6];
  audioMFCC8 = mfccCoeffs[7];
  
  samplesRead = 0;
}

// -----------------------------------------------------------------------------
// CALCULATE VIBRATION FEATURES (REDUCED - 5 features)
// -----------------------------------------------------------------------------
void calculateVibrationFeatures() {
  float x, y, z;
  if (IMU.accelerationAvailable()) {
    IMU.readAcceleration(x, y, z);
    
    // Core features (KEPT)
    vibMagnitude = sqrt(x * x + y * y + z * z);
    vibMagnitudeFeature = vibMagnitude;
    vibRMS_X = abs(x);
    vibRMS_Y = abs(y);
    vibRMS_Z = abs(z);
    
    // Store in history for StdDev calculation
    vibMagnitudeHistory[historyIndex] = vibMagnitudeFeature;
    
    // Calculate StdDev from history (KEPT)
    int count = min(sampleCount + 1, timeSeriesSize);
    
    float sum = 0;
    for (int i = 0; i < count; i++) {
      sum += vibMagnitudeHistory[i];
    }
    float mean = sum / count;
    
    float variance = 0;
    for (int i = 0; i < count; i++) {
      float diff = vibMagnitudeHistory[i] - mean;
      variance += diff * diff;
    }
    vibStdDev = sqrt(variance / count);
    
    // REMOVED: vibMovingAverage, vibPeak, vibFrequencyPeak
  }
}

// -----------------------------------------------------------------------------
// CALCULATE TEMPERATURE FEATURES (REDUCED - 4 features)
// -----------------------------------------------------------------------------
void calculateTemperatureFeatures() {
  static float prevTemp = 0;
  static bool firstRead = true;
  
  if (IMU.temperatureAvailable()) {
    IMU.readTemperature(temperatureDeg);
    tempCurrent = (float)temperatureDeg;
    
    // Store in history
    tempHistory[historyIndex] = tempCurrent;
    
    // Calculate derivative (KEPT)
    if (!firstRead) {
      tempDerivative = tempCurrent - prevTemp;
    } else {
      tempDerivative = 0;
      firstRead = false;
    }
    prevTemp = tempCurrent;
    
    // Calculate stats from history
    int count = min(sampleCount + 1, timeSeriesSize);
    
    float sum = 0;
    for (int i = 0; i < count; i++) {
      sum += tempHistory[i];
    }
    tempMovingAverage = sum / count;  // KEPT
    
    float variance = 0;
    for (int i = 0; i < count; i++) {
      float diff = tempHistory[i] - tempMovingAverage;
      variance += diff * diff;
    }
    // tempStability = variance / count;  // REMOVED
    tempStdDev = sqrt(variance / count);  // KEPT
  }
}

// READ ALL SENSORS AND CALCULATE ALL FEATURES
// -----------------------------------------------------------------------------
void readAllSensorsAndFeatures() {
  calculateAudioFeatures();
  calculateVibrationFeatures();
  calculateTemperatureFeatures();
  
  // Update history index AFTER calculations
  historyIndex = (historyIndex + 1) % timeSeriesSize;
  if (sampleCount < timeSeriesSize) sampleCount++;
}

// -----------------------------------------------------------------------------
// GET TIMESTAMP (YYYY-MM-DD HH:MM:SS)
// -----------------------------------------------------------------------------
String getTimestamp() {
  timeClient.update();
  unsigned long epoch = timeClient.getEpochTime();

  unsigned long SECS_PER_DAY = 86400UL;
  unsigned long SECS_PER_HOUR = 3600UL;
  unsigned long SECS_PER_MIN = 60UL;

  unsigned long days = epoch / SECS_PER_DAY;
  unsigned long remaining = epoch % SECS_PER_DAY;

  int hour = remaining / SECS_PER_HOUR;
  remaining %= SECS_PER_HOUR;

  int minute = remaining / SECS_PER_MIN;
  int second = remaining % SECS_PER_MIN;

  int year = 1970;
  while (true) {
    bool leap = (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0);
    int daysInYear = leap ? 366 : 365;

    if (days >= daysInYear) {
      days -= daysInYear;
      year++;
    } else {
      break;
    }
  }

  int monthDays[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
  bool leapThisYear = (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0);
  if (leapThisYear) monthDays[1] = 29;

  int month = 0;
  while (days >= monthDays[month]) {
    days -= monthDays[month];
    month++;
  }

  int day = days + 1;
  int realMonth = month + 1;

  char buffer[30];
  sprintf(buffer, "%04d-%02d-%02d %02d:%02d:%02d",
          year, realMonth, day, hour, minute, second);

  return String(buffer);
}

// -----------------------------------------------------------------------------
// ENSURE CONNECTION
// -----------------------------------------------------------------------------
bool ensureConnection() {
  if (client.connected()) {
    return true;
  }
  return client.connect(host.c_str(), 443);
}

// -----------------------------------------------------------------------------
// SEND REQUEST (reuses connection)
// -----------------------------------------------------------------------------
void sendRequest(String path, String jsonData) {
  String req =
    "POST " + path + " HTTP/1.1\r\n"
    "Host: " + host + "\r\n"
    "Connection: keep-alive\r\n"
    "Content-Type: application/json\r\n"
    "Content-Length: " + String(jsonData.length()) + "\r\n\r\n" +
    jsonData;

  client.print(req);
  
  unsigned long timeout = millis() + 500;
  while (millis() < timeout) {
    while (client.available()) {
      client.read();
    }
    if (!client.available()) break;
  }
}

// -----------------------------------------------------------------------------
// SETUP
// -----------------------------------------------------------------------------
void setup() {
  if (Serial) Serial.begin(115200);
  delay(500);

  host = projectID + "-default-rtdb.firebaseio.com";

  WiFi.begin(ssid, pass);
  unsigned long startAttempt = millis();

  while (WiFi.status() != WL_CONNECTED) {
    if (millis() - startAttempt > 10000) {
      if (Serial) Serial.println("WiFi failed! Rebooting...");
      delay(2000);
      NVIC_SystemReset();
    }
    delay(500);
  }
  if (Serial) Serial.println("WiFi Connected!");

  timeClient.begin();
  if (Serial) Serial.print("Syncing time...");

  while (!timeClient.update()) {
    timeClient.forceUpdate();
    if (Serial) Serial.print(".");
    delay(500);
  }
  if (Serial) Serial.println("\nTime synced!");

  if (!IMU.begin()) {
   if (Serial) Serial.println("Failed to init IMU!");
   while (1);
  }

  if (!PDM.begin(1, 16000)) {
    if (Serial) Serial.println("Failed to start mic!");
    while (1);
  }
  
  PDM.setBufferSize(sampleBufferSize * sizeof(short));
  PDM.onReceive(onPDMData);

  if (Serial) Serial.println("System Ready! (Industrial-Grade 16 Audio Features)");
}

// -----------------------------------------------------------------------------
// LOOP
// -----------------------------------------------------------------------------
void loop() {
  unsigned long startTime = millis();
  
  readAllSensorsAndFeatures();

  if (Serial) {
    Serial.print("TEMP: ");
    Serial.print(temperatureDeg);
    Serial.print("C | VIB: ");
    Serial.print(vibMagnitude, 4);
    Serial.print("g | MIC: ");
    Serial.print(micLevel);
    Serial.print(" | audioRMS: ");
    Serial.print(audioRMS, 6);
    Serial.print(" | tempStdDev: ");
    Serial.println(tempStdDev, 4);
  }

  sendAllDataToFirebase();

  unsigned long elapsed = millis() - startTime;
  if (Serial) {
    Serial.print("Cycle time: ");
    Serial.print(elapsed);
    Serial.println("ms\n");
  }
  
  if (elapsed < 1000) {
    delay(1000 - elapsed);
  }
}
